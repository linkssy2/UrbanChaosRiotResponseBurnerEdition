cmake_minimum_required(VERSION 3.20)
project(uc_native_texture LANGUAGES C)
if(NOT TARGET uc_native_archive)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../archive_reader" "${CMAKE_CURRENT_BINARY_DIR}/archive_reader")
endif()
add_library(uc_native_texture STATIC texture_asset.c)
target_include_directories(uc_native_texture PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(uc_native_texture PUBLIC uc_native_archive)
set_property(TARGET uc_native_texture PROPERTY C_STANDARD 11)
if(MSVC)
    target_compile_options(uc_native_texture PRIVATE /W4 /WX /O2)
endif()
